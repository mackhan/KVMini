//! TLS 证书模块 — 自签证书生成、PEM I/O、SHA-256 指纹
use crate::error::{KvmError, KvmResult};
use rcgen::{CertificateParams, DistinguishedName, DnType, KeyPair};
use sha2::{Digest, Sha256};

pub struct SelfSignedCert {
    pub cert_der: Vec<u8>,
    pub key_der: Vec<u8>,
}

pub struct CertPem { pub cert_pem: String, pub key_pem: String }

impl SelfSignedCert {
    pub fn generate(common_name: &str) -> KvmResult<Self> {
        let mut params = CertificateParams::new(vec![common_name.to_string()])
            .map_err(|e| KvmError::Internal(format!("cert params: {e}")))?;
        let mut dn = DistinguishedName::new();
        dn.push(DnType::CommonName, common_name);
        params.distinguished_name = dn;
        // 有效期 10 年
        let key = KeyPair::generate()
            .map_err(|e| KvmError::Internal(format!("keygen: {e}")))?;
        let cert = params.self_signed(&key)
            .map_err(|e| KvmError::Internal(format!("sign: {e}")))?;
        Ok(Self { cert_der: cert.der().to_vec(), key_der: key.serialize_der() })
    }

    pub fn fingerprint_sha256_hex(&self) -> String {
        let mut hasher = Sha256::new();
        hasher.update(&self.cert_der);
        hex::encode(hasher.finalize())
    }

    pub fn to_pem(&self) -> CertPem { /* ... */ }
    pub fn from_pem(cert_pem: &str, key_pem: &str) -> KvmResult<Self> { /* ... */ }
}
```

//! TLS 证书持久化 — macOS Keychain / Windows Credential Manager / Linux Secret Service
//! 通过 `keyring` crate 跨平台抽象
use keyring::Entry;
use kvm_core::tls::SelfSignedCert;

const SERVICE: &str = "com.kvmini.app.tls";
const KEY_CERT: &str = "cert_pem";
const KEY_PRIV: &str = "key_pem";

pub fn save(cert: &SelfSignedCert) -> anyhow::Result<()> {
    let pem = cert.to_pem();
    Entry::new(SERVICE, KEY_CERT)?.set_password(&pem.cert_pem)?;
    Entry::new(SERVICE, KEY_PRIV)?.set_password(&pem.key_pem)?;
    Ok(())
}

pub fn load() -> anyhow::Result<SelfSignedCert> {
    let cert_pem = Entry::new(SERVICE, KEY_CERT)?.get_password()?;
    let key_pem = Entry::new(SERVICE, KEY_PRIV)?.get_password()?;
    Ok(SelfSignedCert::from_pem(&cert_pem, &key_pem)?)
}

pub fn load_or_generate(common_name: &str) -> anyhow::Result<SelfSignedCert> {
    if let Ok(c) = load() { return Ok(c); }
    let c = SelfSignedCert::generate(common_name)?;
    save(&c)?;
    Ok(c)
}

pub fn delete() -> anyhow::Result<()> {
    let _ = Entry::new(SERVICE, KEY_CERT)?.delete_credential();
    let _ = Entry::new(SERVICE, KEY_PRIV)?.delete_credential();
    Ok(())
}
```

## 已知风险与缓解

| 风险 | 影响 | 缓解 |
|---|---|---|
| `keyring` crate 在 Linux 无头环境不可用 | Linux 用户无法存证书 | Fallback 到文件存储（chmod 600） |
| macOS 首次调 Keychain 弹授权框 | 用户体验打断 | 启动时异步调，失败降级到文件 |
| Windows Credential Manager 需要用户登录会话 | 服务模式不可用 | 本项目是 GUI App，不考虑服务模式 |
| rustls 0.23 API 变化大，rcgen 0.13 API 也新 | 集成复杂 | 各自先在独立 module 验证，再集成 |
| 不向后兼容导致老用户升级后连不上 | 用户投诉 | changelog/弹窗提示「请两端同时升级到 0.1.38+」 |
| 组 C/D/E 工作量超预期 | 本轮发不出来 | 组 C/D 先做 UI 层「假 TLS 下线 + 加诚实文案」作为 plan B |
